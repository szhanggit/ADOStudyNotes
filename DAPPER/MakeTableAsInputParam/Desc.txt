	[CompanyName] [varchar](50) NOT NULL,

IF EXISTS ( SELECT  1
            FROM    dbo.sysobjects
            WHERE   id = OBJECT_ID(N'[dbo].[spGetCompaniesOfCustomers]')
                    AND OBJECTPROPERTY(id, N'IsProcedure') = 1 )
    DROP PROCEDURE [dbo].[spGetCompaniesOfCustomers]
GO

CREATE PROCEDURE [dbo].[spGetCompaniesOfCustomers]
    @TableName varchar(50) = 'Customers',
	@Customers Customer_Input Readonly
AS
BEGIN
	SET NOCOUNT ON;
    DECLARE @sql nvarchar(3000);
    SET @sql = 'Select CS.CompanyName, CS.ContactTitle, CS.ContactName from '+@TableName+' As CS With(NOLOCK) INNER JOIN @Customers As C On C.CustomerId = CS.CustomerId'
	exec sp_executesql  @sql, N'@Customers Customer_Input Readonly', @Customers;	
END
GO
