            {
                var _list = conn.MO_Query("SELECT MasterVoucherId, [Status] as ComboStatus FROM dbo.VoucherCombo WITH(NOLOCK) WHERE ChildVoucherId = @id", new { Id = ChildVoucherId }).ToList();

                foreach (var obj in _list)
                {
                    long MasterVoucherId = 0;
                    int ComboStatus = 0;

                    if (CommonHelper.PropertyExists(obj, "MasterVoucherId"))
                    {
                        MasterVoucherId = obj.MasterVoucherId;
                    }

                    if (CommonHelper.PropertyExists(obj, "ComboStatus"))
                    {
                        ComboStatus = obj.ComboStatus;
                    }

                    if (MasterVoucherId > 0 && ComboStatus > 0)
                    {
                        _dic.Add(MasterVoucherId, ComboStatus);
                    }
                }
            }

            return _dic;


        public static bool PropertyExists(dynamic obj, string name)
        {
            if (obj == null) return false;
            if (obj is IDictionary<string, object> dict)
            {
                return dict.ContainsKey(name);
            }
            return obj.GetType().GetProperty(name) != null;
        }
